CREATE TABLE `lessons` (
  `lesson_id` int NOT NULL AUTO_INCREMENT,
  `course_id` int DEFAULT NULL COMMENT '所属课程ID',
  `title` varchar(255) DEFAULT NULL COMMENT '课时标题',
  `intro_text` mediumtext COMMENT '课程介绍(富文本或Markdown)',
  `homework_text` mediumtext COMMENT '课后任务(富文本或Markdown)',
  `video_url` varchar(255) DEFAULT NULL COMMENT '视频地址',
  `duration` int unsigned DEFAULT NULL COMMENT '视频时长(秒)',
  `materials_json` json DEFAULT NULL COMMENT '课程资料数组[{name,url}]',
  `code_timeline_json` longtext COMMENT '视频时间轴代码快照(JSON 压缩字符串)',
  `sort_order` smallint unsigned DEFAULT '0' COMMENT '排序序号',
  PRIMARY KEY (`lesson_id`),
  KEY `lessons_ibfk_1` (`course_id`),
  CONSTRAINT `lessons_ibfk_1` FOREIGN KEY (`course_id`) REFERENCES `courses` (`course_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci
